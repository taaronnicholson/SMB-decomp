; Analysis hints for tools/decomp.py
;   entry <bank|f> <addr> [curbank]   extra code entry point
;   words <bank|f> <addr> <n> [bank]  table of n code pointers (lo,hi)
;   lohi  <bank|f> <lo> <hi> <n> [bank|-] [adj]  split lo/hi pointer tables
;   name  <bank|f> <addr> <label>     rename a label
;   ram   <addr> <name>               name a RAM location
words f C825 6                ; NMI state dispatch (indexed by nmi_state, $00)

name f C5C6 Reset
name f C7E3 NMI
name f C84A IRQ
name f C90D SwitchBank        ; A = PRG bank 0-6 -> $8000; shadows in cur_bank
name f C917 SwitchBankFromAC  ; bank taken from $AC
name f C901 SetPPUCtrl        ; writes A to $2000 and shadow
name f C907 SetPPUMask        ; writes A to $2001 and shadow
name f C93D ClearOAMBuffer    ; fills $0200-$02FF with $FF/0
name f CEEE FarCall           ; A=hi X=lo Y=bank : call $AAXX in bank Y, restores bank

ram 0003 ppumask_shadow

ram 0005 tmp_ptr_hi
ram 0012 buttons              ; current controller state (A,B,Sel,Start,U,D,L,R bits)
ram 0013 buttons_prev
ram 0014 buttons_pressed      ; newly pressed this frame
ram 0015 buttons_released

name 0 8205 ReadController
name 0 8245 ClearRAM           ; zeroes RAM $0000-$06FF except the $0100 stack page
dwords 0 8263 6                ; table of 6 pointers to record lists

name 6 8003 Sound_Update       ; per-frame sound engine tick (bank 6 = sound engine)

ram 0052 ptr_hi
ram 0093 stage_index             ; index into the stage record pointer table (bank 0 $8263)
ram 072C game_state            ; index for DispatchGameState table

; ---- bank 5 = entity movement; bank 0 holds a sine table ----
refbank BE8A 0
name 0 BE8A SineTable          ; 256 signed entries, amplitude +-64, index = angle (256 = full turn)
name f CEA6 SinLookup          ; Y = angle -> A = sin (via bank 0 table)
name f CEBA CosLookup          ; Y = angle -> A = cos (= table[angle+$40])
name f C919 SwitchBankKeepX    ; like SwitchBank but preserves X
name f FCDC ScaledMultiply     ; radius x sin/cos helper used by polar movement
name 5 8000 SetEntityPolarPos  ; X=entity: pos = centre(zp_55/56) +- radius*cos/sin(angle)
name 5 809E EntityDeltaFromCentre
ram 04B0 ent_y_lo
ram 04E0 ent_y_hi
ram 0510 ent_x_lo
ram 0540 ent_x_hi
ram 05D0 ent_angle
ram 0600 ent_radius
ram 0660 ent_dy
ram 0690 ent_dx
ram 0570 ent_flags            ; bit tests (e.g. AND #$04) seen in bank 5 movement code

; ---- stage records (bank 0): 6 records, pointer table at $8263 ----

; area = destination area number (0-23, see AreaHeader*); >= $E0 means "go to stage (code-$E0)"; entries are indexed 6*n from record+7
name 0 826F StageRecord0
name 0 82B8 StageRecord1
name 0 8301 StageRecord2
name 0 834A StageRecord3
name 0 8393 StageRecord4

struct 0 83AC 1 wwbbb
ram 0060 player_y_lo          ; loaded from the stage-record spawn entry (y first, then x)
ram 0061 player_y_hi

ram 005F player_x_hi

ram 009A entry_list_lo        ; pointer to the current record's entry list (record+7)
ram 009B entry_list_hi

; ---- PPU update buffer (flushed by NMI) ----
name f C0A8 FlushPPUBuffer     ; reads commands at ppu_buffer: [addr_hi][addr_lo][len][data...], hi=0 ends
ram 0300 ppu_buffer            ; len=1: single byte; len bit7 set: horizontal (+1) else vertical (+32)
name 0 8108 PaletteFadeStep    ; fades the palette at $3F00 toward/away from black (zp_10 = step 0-4, zp_11 = direction)
ram 0010 fade_step
ram 0011 fade_dir

; ---- LZ-style decompressor (graphics and stage data) ----
name f FDEF Decompress         ; ptr_lo/hi = source, $55/$56 = destination; carry set = RAM target, clear = VRAM
name f FDE9 DecompressAndFinish
name f FD84 Decomp_PutByte     ; write A to the destination (RAM via ($55),y, or PPU)
name f FD9F Decomp_ReadOut     ; read back a byte already written (for copy matches)
name f FDBB Decomp_ReadBack
name f FDD7 Decomp_NextSrc
name f FDE0 Decomp_NextDst
ram 0397 decomp_to_ram         ; bit 7 set = output to RAM instead of VRAM

; ---- areas and maps ----
name f D05C LoadArea            ; A = area number (0-23): sets map bank, pointer, width, height
name f FC26 GetMapTile          ; returns the tile id at (ptr_lo/hi = x, $55/$56 = y), applying collected-item flags
name f EF24 MapPtrFromPos       ; map address = map_ptr + (x>>4) + (y>>4) * map_width
ram 00AB current_area
ram 00AC map_bank
ram 0025 map_ptr_lo
ram 0026 map_ptr_hi
ram 0033 map_log2_width         ; width = 8 << this
ram 0023 map_width              ; tiles per row: 16, 32, 64 or 128
ram 0024 map_height             ; rows
ram 03A4 tile_props             ; per-tile property table (bit 0 = solid), 120 bytes decompressed per stage
name f CE3B PlaySong             ; A = song number: switches to bank 6 and calls Sound_PlaySong
name f CE52 SilenceSound         ; calls Sound_Silence in bank 6
ram 000F palette_set             ; area header byte 5: index into the palette fade tables (bank 0 $8040/$8049)
ram 001A tileset_id              ; area header byte 11: selects the tileset record at $8C9B + 8*id

; ---- collectibles (diamonds etc.): tile ids $80-$BB, collected state kept in two bitmaps ----
name f F6D1 GetCollectibleBit   ; map position -> Y = byte, X = bit in collected_flags (via the area's collectible table)
name f F679 ApplyCollectedFlag  ; GetMapTile helper: returns 0 instead of the tile once its flag is set
name f DBB7 FilterCollectedTile ; tile ids $80-$BB: hide if that kind's bit in collected_by_kind is set
ram 009D collected_by_kind      ; 8-byte bitmap, one bit per collectible tile kind ($80 + n)
ram 0100 collected_flags        ; bitmap, one bit per collectible placement (flag index from the collectible table)
ram 0480 ent_type                ; spawn record byte 0 (entity kind)
ram 05A0 ent_param               ; spawn record byte 6
ram 0630 ent_param2              ; spawn record byte 9

; ---- credits scroller (bank 5 holds the credits text; $C2/$C3 walk it) ----
name f CF1E CreditsStep         ; prints the next credits line (tile = char + $94), $FE = end
ram 00C2 credits_ptr_lo
ram 00C3 credits_ptr_hi

; ---- entity drawing (bank 1) ----
name 1 8A5B DrawEntity          ; X = slot: builds OAM sprites from the entity's animation group/frame
name 1 8A52 MarkEntityOffscreen ; sets ent_flags bit 2 when the entity is outside the screen window
ram 0120 anim_tile_base         ; per animation group: first CHR tile number of its sprite sheet (loaded per stage)
